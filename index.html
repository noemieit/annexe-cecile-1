// ── Paramètres réels Cécile ──

const CAP_APPORTE  = 500000;

const LOMBARD_INIT = 290000;

const RATE_LOMBARD = 0.0325;

const PE_INIT      = 200000;

const LIQUIDE_INIT = 590000;

const RATE_LIQUIDE = 0.082;

const RATE_PE      = 0.13;


const fmt = v => new Intl.NumberFormat('fr-FR',{style:'currency',currency:'EUR',maximumFractionDigits:0}).format(v);

const fmtpct = v => v.toFixed(1)+' %';


function calcAn(yr){

  const liquide_brut = LIQUIDE_INIT * Math.pow(1 + RATE_LIQUIDE, yr);

  const pe_val       = PE_INIT * Math.pow(1 + RATE_PE, yr);

  const lombard_du   = LOMBARD_INIT * Math.pow(1 + RATE_LOMBARD, yr);

  const valeur_totale = liquide_brut + pe_val;

  const gains_totaux = valeur_totale - CAP_APPORTE;

  const qp_gain = gains_totaux > 0 ? gains_totaux / valeur_totale : 0;

  const montant_imposable = liquide_brut * qp_gain;

  const avant8 = yr < 8;

  let fisc;

  if(avant8){

    fisc = montant_imposable * 0.30;

  } else {

    const imposable_ir = Math.max(0, montant_imposable - 4600);

    fisc = imposable_ir * 0.075 + montant_imposable * 0.172;

  }

  const net = liquide_brut - lombard_du - fisc;

  return {liquide_brut, pe_val, lombard_du, valeur_totale, gains_totaux, qp_gain, montant_imposable, fisc, net, avant8};

}
